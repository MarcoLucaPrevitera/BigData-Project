
-----------------------------------------ARTICLES----------------------------------------

#LOAD ARTICLES
LOAD CSV WITH HEADERS FROM 'file:///article.csv' AS row
WITH row
MERGE (art:Article {id: row.key})
ON CREATE SET art.date = row['date'], art.title = row['title0'], art.ee = row['ee0'], art.note = row['note0'], art.publisher = row['publisher0']

#LOAD AUTHORS
LOAD CSV WITH HEADERS FROM 'file:///article.csv' AS row
UNWIND range(0,21) AS i
WITH i, row WHERE row['author'+toString(i)] IS NOT NULL
MERGE (aut:Author {name: row['author'+toString(i)]})

#LOAD WRITTEN_BY RELATIONS
LOAD CSV WITH HEADERS FROM 'file:///article.csv' AS row
UNWIND range(0,21) AS i
WITH i, row WHERE row['author'+toString(i)] IS NOT NULL
MATCH (aut:Author {name: row['author'+toString(i)]})
MATCH (art:Article {id: row.key})
MERGE (art)-[:WRITE]-(aut)

#LOAD JOURNALS AND RELATIONS
LOAD CSV WITH HEADERS FROM 'file:///article.csv' AS row
WITH row WHERE row['journal0'] IS NOT NULL
MERGE (j:Journal {name: row['journal0']})
WITH j, row
MATCH (art:Article {id: row.key})
MERGE (art)-[rel:PUBLICATE]-(j)
ON CREATE SET rel.year = row.year0, rel.month = row.month0, rel.volume = row.volume0, rel.pages = row.pages0, rel.number = row.number0

#--------------------------------ALL IN ONE ARTICLE------------------------------------------------------
LOAD CSV WITH HEADERS FROM 'file:///article.csv' AS row
WITH row
MERGE (art:Article {id: row.key})
ON CREATE SET art.date = row['date'], art.title = row['title0'], art.ee = row['ee0'], art.note = row['note0'], art.publisher = row['publisher0']
WITH row
UNWIND range(0,21) AS i
WITH i, row WHERE row['author'+toString(i)] IS NOT NULL
MERGE (aut:Author {name: row['author'+toString(i)]})
WITH i, row
MATCH (art:Article {id: row.key})
MERGE (art)-[:WRITE]-(aut)
WITH row WHERE row['journal0'] IS NOT NULL
MERGE (j:Journal {name: row['journal0']})
WITH j, row
MATCH (art:Article {id: row.key})
MERGE (art)-[rel:PUBLICATE]-(j)
ON CREATE SET rel.year = row.year0, rel.month = row.month0, rel.volume = row.volume0, rel.pages = row.pages0, rel.number = row.number0

#LOAD MASTERTHESIS
LOAD CSV WITH HEADERS FROM 'file:///mastersthesis.csv' AS row
WITH row
MERGE (mast:Mastersthesis {id: row.key})
ON CREATE SET mast.date = row['date'], mast.title = row['title0'], mast.ee = row['ee0'], mast.note = row['note0'], mast.school = row['school0'], mast.year = row['year0']

#LOAD AUTHORS
LOAD CSV WITH HEADERS FROM 'file:///mastersthesis.csv' AS row
WITH row WHERE row['author0'] IS NOT NULL
MERGE (aut:Author {name: row['author0']})
WITH row, aut
MATCH (mast:Mastersthesis {id: row.key})
MERGE (mast)-[:WRITE]-(aut)
