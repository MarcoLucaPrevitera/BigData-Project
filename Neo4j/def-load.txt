-----------------------------------------ARTICLES----------------------------------------
#LOAD ARTICLES
LOAD CSV WITH HEADERS FROM 'file:///article.csv' AS row
WITH row
MERGE (art:Article:Publication {id: row.key})
ON CREATE SET art.date = row['date'], art.title = row['title0'], art.ee = row['ee0'], art.note = row['note0'], art.publisher = row['publisher0']

#LOAD AUTHORS
LOAD CSV WITH HEADERS FROM 'file:///article.csv' AS row
WITH row
MATCH (article:Article {id: row.key})
UNWIND range(0,21) AS i
WITH i, article, row WHERE row['author'+toString(i)] IS NOT NULL
MERGE (aut:Author {name: row['author'+toString(i)]})
MERGE (article)-[:WRITTEN_BY]->(aut)

#LOAD JOURNALS
LOAD CSV WITH HEADERS FROM 'file:///article.csv' AS row
WITH row
MATCH (art:Article {id: row.key})
WITH art, row WHERE row['journal0'] IS NOT NULL
MERGE (j:Journal {name: row['journal0']})
MERGE (art)-[rel:PUBLICATED_BY]->(j)
ON CREATE SET rel.year = row.year0, rel.month = row.month0, rel.volume = row.volume0, rel.pages = row.pages0, rel.number = row.number0

#LOAD AUTHORS
LOAD CSV WITH HEADERS FROM 'file:///mastersthesis.csv' AS row
WITH row WHERE row['author0'] IS NOT NULL
MERGE (aut:Author {name: row['author0']})
WITH row, aut
MATCH (mast:Mastersthesis {id: row.key})
MERGE (mast)-[:WRITTEN_BY]->(aut)
