Python Supports JSON Natively!
"import json"
	from file.json

-----------SERIALIZATION-----------------------
data = {
    "president": {
        "name": "Zaphod Beeblebrox",
        "species": "Betelgeusian"
    }
}

with open("data_file.json", "w") as write_file:
    json.dump(data, write_file)
------------------------------------------------


--------DESERIALIZATION--------------------------
with open("data_file.json", "r") as read_file:
    data = json.load(read_file)
-------------------------------------------------
